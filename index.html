const allNavLinks = document.querySelectorAll('.nav-link');
const sectionIds = [...new Set(Array.from(allNavLinks).map(l => l.getAttribute('href')))];
const spySections = sectionIds.map(id => document.querySelector(id)).filter(Boolean);
const spy = new IntersectionObserver((entries) => {
entries.forEach(entry => {
if (entry.isIntersecting) {
const id = entry.target.getAttribute('id');
allNavLinks.forEach(l => l.classList.toggle('active', l.getAttribute('href') === '#' + id));
}
});
}, { rootMargin: '-40% 0px -55% 0px' });
spySections.forEach(s => spy.observe(s));
// Reveal on scroll
const reveals = document.querySelectorAll('.reveal');
const revealer = new IntersectionObserver((entries) => {
entries.forEach(entry => {
if (entry.isIntersecting) {
entry.target.classList.add('is-visible');
revealer.unobserve(entry.target);
}
});
}, { threshold: 0.08 });
reveals.forEach(r => revealer.observe(r));
// Filter tabs
document.querySelectorAll('.filter-group').forEach(group => {
const buttons = group.querySelectorAll('.pub-filter button');
const items = group.querySelectorAll('.pub-list li');
function applyFilter(filter) {
items.forEach(item => {
item.style.display = item.dataset.category === filter ? '' : 'none';
});
}
const initialTab = group.querySelector('.pub-filter .is-current');
if (initialTab) applyFilter(initialTab.dataset.filter);
buttons.forEach(button => {
button.addEventListener('click', () => {
buttons.forEach(b => b.classList.remove('is-current'));
button.classList.add('is-current');
applyFilter(button.dataset.filter);
});
});
});
